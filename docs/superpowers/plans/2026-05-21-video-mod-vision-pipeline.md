# Warframe 视频 MOD 视觉解析系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立一个可审计的半自动视频解析流水线，用 Playwright/视频帧、OCR、ImageHash/OpenCV、YOLO 和人工确认辅助解析 B 站 Warframe 配卡视频中的武器、MOD、赋能、灵化和流派信息。

**Architecture:** 先做离线维护工具，不接入用户回答路径：采集公开视频帧，定位候选区域，OCR 读文字，图标匹配给候选，VLM 只做画面类型说明，最后输出 `needs_review` 的解析草稿供人工确认。确认后的结果再进入结构化知识库或 B 站推荐库，避免未验证视觉识别污染 Agent 回答。

**Tech Stack:** Python 3.14、Playwright、OpenCV、ImageHash、PaddleOCR、可选 Ultralytics YOLO、可选本地/云端 VLM、pytest、JSON/SQLite。

from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Any


_BVID_RE = re.compile(r"BV[0-9A-Za-z]+")


@dataclass(frozen=True)
class VideoSource:
    url: str
    bvid: str
    platform: str = "bilibili"

    @classmethod
    def from_url(cls, url: str) -> "VideoSource":
        match = _BVID_RE.search(str(url or ""))
        if not match:
            raise ValueError("Bilibili URL must contain a BVID")
        bvid = match.group(0)
        return cls(url=f"https://www.bilibili.com/video/{bvid}", bvid=bvid)


@dataclass(frozen=True)
class FrameSample:
    path: str
    timestamp_seconds: float
    width: int = 0
    height: int = 0


@dataclass(frozen=True)
class DetectedRegion:
    kind: str
    box: list[int]
    confidence: float
    frame_path: str = ""


@dataclass(frozen=True)
class OcrCandidate:
    text: str
    region_kind: str
    confidence: float
    frame_path: str = ""


@dataclass(frozen=True)
class IconMatch:
    label: str
    kind: str
    score: float
    source_icon: str
    frame_path: str = ""
    region_kind: str = ""


@dataclass(frozen=True)
class ParsedBuildDraft:
    source: VideoSource
    title: str
    frames: list[FrameSample] = field(default_factory=list)
    regions: list[DetectedRegion] = field(default_factory=list)
    ocr_candidates: list[OcrCandidate] = field(default_factory=list)
    icon_matches: list[IconMatch] = field(default_factory=list)
    inferred_weapon: str = ""
    inferred_category: str = ""
    inferred_archetype: str = ""
    notes: list[str] = field(default_factory=list)
    needs_review: bool = True
    trusted_for_agent_answers: bool = False

    def to_dict(self) -> dict[str, Any]:
        return {
            "source": {"url": self.source.url, "bvid": self.source.bvid, "platform": self.source.platform},
            "title": self.title,
            "frames": [frame.__dict__ for frame in self.frames],
            "regions": [region.__dict__ for region in self.regions],
            "ocr_candidates": [candidate.__dict__ for candidate in self.ocr_candidates],
            "icon_matches": [match.__dict__ for match in self.icon_matches],
            "inferred_weapon": self.inferred_weapon,
            "inferred_category": self.inferred_category,
            "inferred_archetype": self.inferred_archetype,
            "notes": list(self.notes),
            "needs_review": self.needs_review,
            "trusted_for_agent_answers": self.trusted_for_agent_answers,
        }
```

from __future__ import annotations

from .models import DetectedRegion, FrameSample


_DEFAULT_REGION_RATIOS = {
    "weapon_name": [0.05, 0.05, 0.35, 0.15],
    "mod_grid": [0.18, 0.22, 0.82, 0.78],
    "arcane_slots": [0.78, 0.18, 0.96, 0.42],
    "exilus_slot": [0.04, 0.28, 0.18, 0.48],
    "incarnon_choices": [0.12, 0.12, 0.88, 0.88],
}


class FixedRegionLocator:
    def __init__(self, region_ratios: dict[str, list[float]] | None = None) -> None:
        self.region_ratios = region_ratios or _DEFAULT_REGION_RATIOS

    def locate(self, frame: FrameSample) -> list[DetectedRegion]:
        width = frame.width or 1920
        height = frame.height or 1080
        regions = []
        for kind, ratio_box in self.region_ratios.items():
            x1, y1, x2, y2 = ratio_box
            regions.append(DetectedRegion(
                kind=kind,
                box=[int(width * x1), int(height * y1), int(width * x2), int(height * y2)],
                confidence=0.35,
                frame_path=frame.path,
            ))
        return regions
```

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import imagehash
from PIL import Image

from .models import IconMatch


@dataclass(frozen=True)
class _IconEntry:
    label: str
    kind: str
    path: Path
    phash: imagehash.ImageHash


class IconHashIndex:
    def __init__(self, entries: list[_IconEntry]) -> None:
        self.entries = entries

    @classmethod
    def from_directory(cls, directory: Path, *, kind: str) -> "IconHashIndex":
        entries = []
        for path in sorted(Path(directory).glob("*.png")):
            with Image.open(path) as image:
                entries.append(_IconEntry(
                    label=path.stem,
                    kind=kind,
                    path=path,
                    phash=imagehash.phash(image.convert("RGB")),
                ))
        return cls(entries)

    def match(self, image_path: Path, *, limit: int = 3) -> list[IconMatch]:
        if limit <= 0 or not self.entries:
            return []
        with Image.open(image_path) as image:
            query_hash = imagehash.phash(image.convert("RGB"))
        scored = []
        max_distance = 64
        for entry in self.entries:
            distance = query_hash - entry.phash
            score = max(0.0, 1.0 - (distance / max_distance))
            scored.append(IconMatch(
                label=entry.label,
                kind=entry.kind,
                score=round(score, 4),
                source_icon=str(entry.path),
            ))
        scored.sort(key=lambda match: match.score, reverse=True)
        return scored[:limit]
```

from __future__ import annotations

from pathlib import Path
from typing import Any

from .models import FrameSample, IconMatch, ParsedBuildDraft, VideoSource
from .ocr import OcrEngine
from .regions import FixedRegionLocator


class BuildDraftBuilder:
    def __init__(self, *, region_locator: FixedRegionLocator, ocr_engine: OcrEngine | None, icon_index: Any | None) -> None:
        self.region_locator = region_locator
        self.ocr_engine = ocr_engine
        self.icon_index = icon_index

    def build(self, *, source: VideoSource, title: str, frames: list[FrameSample]) -> ParsedBuildDraft:
        regions = []
        ocr_candidates = []
        icon_matches: list[IconMatch] = []
        for frame in frames:
            frame_regions = self.region_locator.locate(frame)
            regions.extend(frame_regions)
            if self.ocr_engine is not None:
                for region in frame_regions:
                    ocr_candidates.extend(self.ocr_engine.read_region(frame.path, region))
            if self.icon_index is not None and Path(frame.path).exists():
                icon_matches.extend(self.icon_index.match(Path(frame.path), limit=3))
        inferred_weapon = ""
        for candidate in ocr_candidates:
            if candidate.region_kind == "weapon_name" and candidate.text.strip():
                inferred_weapon = candidate.text.strip()
                break
        return ParsedBuildDraft(
            source=source,
            title=title,
            frames=frames,
            regions=regions,
            ocr_candidates=ocr_candidates,
            icon_matches=icon_matches,
            inferred_weapon=inferred_weapon,
            notes=["自动解析草稿，必须人工确认后才能入库。"],
            needs_review=True,
            trusted_for_agent_answers=False,
        )
```

from __future__ import annotations

import argparse
from pathlib import Path

from warframe_agent.video_analysis.draft import BuildDraftBuilder
from warframe_agent.video_analysis.models import FrameSample, VideoSource
from warframe_agent.video_analysis.ocr import FakeOcrEngine
from warframe_agent.video_analysis.regions import FixedRegionLocator
from warframe_agent.video_analysis.storage import VideoParseDraftStore


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Create a human-review draft from a Bilibili Warframe build video.")
    parser.add_argument("url")
    parser.add_argument("--title", default="")
    parser.add_argument("--timestamp", type=float, action="append", default=[])
    parser.add_argument("--frame", action="append", default=[])
    parser.add_argument("--output", default="data/video_parse_drafts.jsonl")
    parser.add_argument("--fake-ocr-weapon", default="")
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    source = VideoSource.from_url(args.url)
    timestamps = args.timestamp or [0.0]
    frame_paths = args.frame or [f"data/video_frames/{source.bvid}-{int(ts)}.png" for ts in timestamps]
    frames = [
        FrameSample(path=path, timestamp_seconds=timestamps[min(index, len(timestamps) - 1)], width=1920, height=1080)
        for index, path in enumerate(frame_paths)
    ]
    ocr = FakeOcrEngine({"weapon_name": [args.fake_ocr_weapon]} if args.fake_ocr_weapon else {})
    builder = BuildDraftBuilder(region_locator=FixedRegionLocator(), ocr_engine=ocr, icon_index=None)
    draft = builder.build(source=source, title=args.title, frames=frames)
    VideoParseDraftStore(Path(args.output)).append(draft)
    print(f"draft written: {args.output}")


if __name__ == "__main__":
    main()

class PaddleOcrEngine:
    def __init__(self, *, lang: str = "ch", enabled: bool = True) -> None:
        self.lang = lang
        self._ocr = None
        if enabled:
            try:
                from paddleocr import PaddleOCR
                self._ocr = PaddleOCR(lang=lang, use_angle_cls=True)
            except Exception:
                self._ocr = None

    @property
    def available(self) -> bool:
        return self._ocr is not None

    def read_region(self, image_path: str, region: DetectedRegion) -> list[OcrCandidate]:
        if self._ocr is None:
            return []
        result = self._ocr.ocr(image_path, cls=True)
        candidates = []
        for page in result or []:
            for item in page or []:
                if len(item) >= 2 and isinstance(item[1], (list, tuple)):
                    text = str(item[1][0]).strip()
                    confidence = float(item[1][1])
                    if text:
                        candidates.append(OcrCandidate(text=text, region_kind=region.kind, confidence=confidence, frame_path=image_path))
        return candidates
```
